async function addBatch(drafts,batch){
const repo="shutantipura/hsk5-flashcards",branch="reading-library-200";
const [v,b,e]=await Promise.all([tools.mcp__GitHub__fetch_file({repository_full_name:repo,path:"data/vocab.json"}),tools.mcp__GitHub__fetch_blob({repository_full_name:repo,blob_sha:"f03b3622d740b336b55b3c40695faa2ce9716d1a"}),tools.mcp__GitHub__fetch_file({repository_full_name:repo,path:"js/coverage-practice-data.js",ref:branch})]);
const vocab=JSON.parse(v.result.content),parse=s=>JSON.parse(s.slice(s.indexOf("["),s.lastIndexOf("]")+1)),old=parse(b.result.content),extra=parse(e.result.content),levels=new Map(),exact=new Map();for(const w of vocab){levels.set(w.h,Math.min(w.l,levels.get(w.h)??99));if(!exact.has(w.l))exact.set(w.l,new Set());exact.get(w.l).add(w.h);}
const seen=new Set([...old,...extra].filter(q=>q.kind==="reading").flatMap(q=>q.passageTokens));
const byLevel=new Map([4,5].map(l=>[l,new Set([...old,...extra].filter(q=>q.kind==="reading"&&(q.passageLevel||q.level)===l).flatMap(q=>q.passageTokens))]));
function seg(s,l){const out=[];for(const part of s.split(/[\p{P}\p{Z}\s]+/u).filter(Boolean)){const a=Array.from(part),dp=Array(a.length+1).fill(null);dp[a.length]={cost:0,t:[]};for(let i=a.length-1;i>=0;i--){for(let j=i+1;j<=Math.min(a.length,i+12);j++){const w=a.slice(i,j).join("");if(!(levels.get(w)<=l)||!dp[j])continue;const cost=dp[j].cost+1-(exact.get(l).has(w)?.04:0);if(!dp[i]||dp[i].cost>cost)dp[i]={cost,t:[w,...dp[j].t]};}}if(!dp[0]){let reachable=new Set([0]);for(let i=0;i<a.length;i++)if(reachable.has(i))for(let j=i+1;j<=Math.min(a.length,i+12);j++)if(levels.get(a.slice(i,j).join(""))<=l)reachable.add(j);const k=Math.max(...reachable);return {error:part,blocked:a.slice(Math.max(0,k-2),k+5).join("")};}out.push(...dp[0].t);}return {t:out};}
const add=[],reject=[];for(const [i,d] of drafts.entries()){const [level,title,passage,prompt,answer,other1,other2,translation]=d;const choices=[answer,other1,other2];const p=seg(passage,level),q=seg(prompt,level),c=choices.map(x=>seg(x,level));if(!p.t||!q.t||c.some(x=>!x.t)){reject.push({i,title,p:p.error,blocked:p.blocked,q:q.error,c:c.map(x=>x.error)});continue;}const all=[...new Set(p.t)].filter(t=>exact.get(level).has(t));const novel=all.filter(t=>!seen.has(t)),levelNovel=all.filter(t=>seen.has(t)&&!byLevel.get(level).has(t));const words=[...novel,...levelNovel,...all.filter(t=>byLevel.get(level).has(t))].slice(0,Math.max(5,Math.min(18,novel.length+levelNovel.length)));if(words.length<5||p.t.length<80||p.t.length>120){reject.push({i,title,len:p.t.length,targets:words.length});continue;}const id="r-library-"+batch+"-"+(i+1),k=i%3,opts=choices.slice(k).concat(choices.slice(0,k));if([...old,...extra,...add].some(x=>x.id===id+"-1"))throw Error("duplicate ID "+id);add.push({id:id+"-1",kind:"reading",level,passageId:id,passageLevel:level,title,passage,translation,passageTokens:p.t,words,questionType:"detail",prompt,answer,choices:opts,tokens:p.t.concat(q.t,...opts.map(s=>seg(s,level).t)),explain:"The correct answer follows from the details in the story."});for(const t of p.t){seen.add(t);byLevel.get(level).add(t);}}
text({batch,accepted:add.map(q=>({id:q.id,l:q.level,n:q.passageTokens.length,t:q.words.length})),rejected:reject,counts:[4,5].map(l=>({l,n:new Set([...old,...extra,...add].filter(q=>q.kind==="reading"&&(q.passageLevel||q.level)===l).map(q=>q.passageId)).size}))});
if(add.length){const content="// Additional original reading/listening passages, checked against the bundled syllabus.\nexport const COVERAGE_PRACTICE = [\n"+[...extra,...add].map(q=>JSON.stringify(q)).join(",\n")+"\n];\n";text(await tools.mcp__GitHub__update_file({repository_full_name:repo,path:"js/coverage-practice-data.js",sha:e.result.sha,content,message:"Add validated HSK reading library batch "+batch,branch}));}
}
